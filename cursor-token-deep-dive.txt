#!/usr/bin/env python3
"""
Cursor Token Deep Dive (cursor-token-deep-dive)
Recursively unpacks SQLite ItemTable blobs to mine hidden token metrics and costs.
"""

import os
import sqlite3
import json
import glob

CURSOR_USER_DIR = os.path.expanduser("~/Library/Application Support/Cursor/User")
GLOBAL_DB_PATH = os.path.join(CURSOR_USER_DIR, "globalStorage", "state.vscdb")
WORKSPACE_STORAGE_DIR = os.path.join(CURSOR_USER_DIR, "workspaceStorage")

def search_json_recursively(data, target_keys, path=""):
    """Recursively search a nested JSON structure for target keys."""
    results = {}
    if isinstance(data, dict):
        for k, v in data.items():
            current_path = f"{path}.{k}" if path else k
            if any(tk in k.lower() for tk in target_keys):
                results[current_path] = v
            if isinstance(v, (dict, list)):
                results.update(search_json_recursively(v, target_keys, current_path))
    elif isinstance(data, list):
        for idx, item in enumerate(data):
            current_path = f"{path}[{idx}]"
            results.update(search_json_recursively(item, target_keys, current_path))
    return results

def inspect_db(db_path, scope_name):
    if not os.path.exists(db_path):
        return
    
    try:
        conn = sqlite3.connect(db_path)
        cursor = conn.cursor()
        
        # Check available tables
        cursor.execute("SELECT name FROM sqlite_master WHERE type='table';")
        tables = [t[0] for t in cursor.fetchall()]
        table_name = "ItemTable" if "ItemTable" in tables else ("cursorDiskKV" if "cursorDiskKV" in tables else None)
        
        if not table_name:
            conn.close()
            return

        cursor.execute(f"SELECT key, value FROM {table_name};")
        rows = cursor.fetchall()
        
        target_tokens_keys = ["token", "cost", "usage", "prompttokens", "completiontokens", "spend"]
        found_hits = 0

        for key, value in rows:
            if not value:
                continue
            
            # Try parsing as JSON
            try:
                parsed = json.loads(value)
                matches = search_json_recursively(parsed, target_tokens_keys)
                if matches:
                    print(f"\n[{scope_name}] Key match -> '{key}'")
                    for m_path, m_val in matches.items():
                        print(f"    {m_path}: {m_val}")
                    found_hits += 1
                    if found_hits >= 5: # Limit per DB to prevent spam
                        break
            except Exception:
                # If it's a string containing target words directly
                if any(tk in value.lower() for tk in target_tokens_keys) and len(value) < 1000:
                    print(f"\n[{scope_name}] Raw string match in key '{key}': {value[:200]}")
                    found_hits += 1
                    
        conn.close()
    except Exception as e:
        pass

if __name__ == "__main__":
    print("Searching Cursor databases for hidden token and cost payloads...")
    inspect_db(GLOBAL_DB_PATH, "GLOBAL")
    
    workspace_dbs = glob.glob(os.path.join(WORKSPACE_STORAGE_DIR, "*", "state.vscdb"))
    for db_path in workspace_dbs:
        ws_id = os.path.basename(os.path.dirname(db_path))
        inspect_db(db_path, f"WORKSPACE-{ws_id[:6]}")
    print("\nDeep dive scan complete.")
