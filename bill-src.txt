#!/bin/bash
VERSION="1.0.0"
BASE_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
DATA_DIR="${BASE_DIR}/data"
CONFIG_DIR="${HOME}/.config/cursor-billing"
LOG_DIR="${BASE_DIR}/logs"
BILLING_FILE="${DATA_DIR}/billing-data.json"
CONFIG_FILE="${CONFIG_DIR}/config.json"
LOG_FILE="${LOG_DIR}/billing.log"
init_environment() {
    mkdir -p "${DATA_DIR}" 2>/dev/null
    mkdir -p "${CONFIG_DIR}" 2>/dev/null
    mkdir -p "${LOG_DIR}" 2>/dev/null
    if [ ! -f "${CONFIG_FILE}" ]; then
        create_default_config
    fi
}
create_default_config() {
    cat > "${CONFIG_FILE}" << 'EOC'
{
  "version": "1.0.0",
  "created": "$(date -Iseconds)",
  "settings": {
    "refresh_interval": 3600,
    "cache_duration": 86400,
    "log_level": "info",
    "security": {
      "encrypt_data": false,
      "require_manual_update": true
    }
  }
}
EOC
}
log_operation() {
    local message="$1"
    local level="${2:-info}"
    local timestamp="$(date -Iseconds 2>/dev/null || date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo "[${timestamp}] [${level}] ${message}" >> "${LOG_FILE}"
}
get_billing_data() {
    if [ -f "${BILLING_FILE}" ]; then
        cat "${BILLING_FILE}" 2>/dev/null || echo "null"
    else
        echo "null"
    fi
}
save_billing_data() {
    local data="$1"
    echo "${data}" > "${BILLING_FILE}"
    chmod 600 "${BILLING_FILE}" 2>/dev/null
    log_operation "Billing data updated"
}
validate_json() {
    local data="$1"
    if echo "${data}" | grep -q "plan" && \
       echo "${data}" | grep -q "amount" && \
       echo "${data}" | grep -q "status"; then
        return 0
    else
        return 1
    fi
}
get_timestamp() {
    date -Iseconds 2>/dev/null || date -u +%Y-%m-%dT%H:%M:%SZ
}
EOF

check_security() {
    echo ""; echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    local dirs=("${DATA_DIR}" "${CONFIG_DIR}" "${LOG_DIR}")
    for dir in "${dirs[@]}"; do
        if [ -d "${dir}" ]; then
            local perms=$(stat -c "%a" "${dir}" 2>/dev/null || stat -f "%Lp" "${dir}" 2>/dev/null)
            if [ "${perms}" = "700" ] || [ "${perms}" = "755" ]; then
                echo "  ✓ ${dir}: ${perms} (secure)"
            else
                echo "  ⚠ ${dir}: ${perms} (recommend: 700)"
            fi
        fi
    done
    if [ -f "${BILLING_FILE}" ]; then
        local perms=$(stat -c "%a" "${BILLING_FILE}" 2>/dev/null || stat -f "%Lp" "${BILLING_FILE}" 2>/dev/null)
        if [ "${perms}" = "600" ]; then
            echo "  ✓ ${BILLING_FILE}: ${perms} (secure)"
        else
            echo "  ⚠ ${BILLING_FILE}: ${perms} (recommend: 600)"
        fi
    fi
    if [ -f "${BILLING_FILE}" ]; then
        if grep -q "password\|secret\|token\|key" "${BILLING_FILE}" 2>/dev/null; then
            echo "  ⚠ WARNING: Sensitive keywords detected"
        else
            echo "  ✓ No sensitive keywords detected"
        fi
    fi
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

    read -p "Requests Limit [5000]: " requests_limit

VERSION="1.0.0"
SCRIPT_NAME="$(basename "$0")"
show_help() {
    cat << EOF
Cursor Billing Tool - Complete Source Bundle
USAGE: ${SCRIPT_NAME} [OPTIONS]
OPTIONS: --extract, -e | --output-dir, -o DIR | --list, -l | --help, -h
EOF
}
list_files() {
    echo "Files: bin/cursor-billing | bin/cursor-billing.ps1 | lib/core.sh | lib/display.sh | lib/security.sh"
    echo "README.md | install.sh | VERSION | examples/example-data.json (Total: 9 files)"
}
extract_files() {
    local output_dir="${1:-.}"
    echo "Extracting files to: ${output_dir}"
    mkdir -p "${output_dir}/bin" "${output_dir}/lib" "${output_dir}/examples"
    local script_content="$(cat "$0")"
    local current_file=""; local in_file=0; local file_content=""
    echo "$script_content" | while IFS= read -r line; do
        if [[ "$line" =~ ^#\ =+\ FILE:\ (.*)$ ]]; then
            if [ $in_file -eq 1 ] && [ -n "$current_file" ]; then
                echo "$file_content" > "${output_dir}/${current_file}"
                echo "  ✓ Extracted: ${current_file}"; file_content=""
            fi
            current_file="${BASH_REMATCH[1]}"; in_file=1; file_content=""
            continue
        fi
        if [ $in_file -eq 1 ]; then
            if [[ "$line" =~ ^cat\ <<\ \'EOF\'\ >\ /dev/null$ ]] || [[ "$line" == "EOF" ]] || [[ "$line" =~ ^#\ =+\$ ]] || [[ -z "$line" && -z "$file_content" ]]; then
                continue
            fi
            if [ -n "$file_content" ]; then
                file_content="${file_content}"$'\n'"${line}"
            else
                file_content="${line}"
            fi
        fi
    done
    if [ $in_file -eq 1 ] && [ -n "$current_file" ] && [ -n "$file_content" ]; then
        echo "$file_content" > "${output_dir}/${current_file}"
        echo "  ✓ Extracted: ${current_file}"
    fi
    chmod +x "${output_dir}/bin/cursor-billing" 2>/dev/null || true
    chmod +x "${output_dir}/install.sh" 2>/dev/null || true
    echo ""; echo "Extraction complete!"
    echo "To use: cd ${output_dir} && ./install.sh"
}
case "$1" in
    --list|-l) list_files ;;
    --extract|-e)
        if [ "$2" = "--output-dir" ] || [ "$2" = "-o" ]; then
            extract_files "$3"
        else
            extract_files "."
        fi
        ;;
    --help|-h) show_help ;;
    *) echo "Unknown option: $1"; echo "Try '${SCRIPT_NAME} --help'"; exit 1 ;;
esac
