#!/bin/bash
# Core library functions
# Cursor Billing Display Tool

VERSION="1.0.0"
BASE_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
DATA_DIR="${BASE_DIR}/data"
CONFIG_DIR="${HOME}/.config/cursor-billing"
LOG_DIR="${BASE_DIR}/logs"
BILLING_FILE="${DATA_DIR}/billing-data.json"
CONFIG_FILE="${CONFIG_DIR}/config.json"
LOG_FILE="${LOG_DIR}/billing.log"

init_environment() {
    mkdir -p "${DATA_DIR}" 2>/dev/null
    mkdir -p "${CONFIG_DIR}" 2>/dev/null
    mkdir -p "${LOG_DIR}" 2>/dev/null
    
    if [ ! -f "${CONFIG_FILE}" ]; then
        create_default_config
    fi
}

create_default_config() {
    cat > "${CONFIG_FILE}" << 'EOC'
{
  "version": "1.0.0",
  "created": "$(date -Iseconds)",
  "settings": {
    "refresh_interval": 3600,
    "cache_duration": 86400,
    "log_level": "info",
    "security": {
      "encrypt_data": false,
      "require_manual_update": true
    }
  }
}
EOC
}

log_operation() {
    local message="$1"
    local level="${2:-info}"
    local timestamp="$(date -Iseconds 2>/dev/null || date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo "[${timestamp}] [${level}] ${message}" >> "${LOG_FILE}"
}

get_billing_data() {
    if [ -f "${BILLING_FILE}" ]; then
        cat "${BILLING_FILE}" 2>/dev/null || echo "null"
    else
        echo "null"
    fi
}

save_billing_data() {
    local data="$1"
    echo "${data}" > "${BILLING_FILE}"
    chmod 600 "${BILLING_FILE}" 2>/dev/null
    log_operation "Billing data updated"
}

validate_json() {
    local data="$1"
    if echo "${data}" | grep -q "plan" && \
       echo "${data}" | grep -q "amount" && \
       echo "${data}" | grep -q "status"; then
        return 0
    else
        return 1
    fi
}

get_timestamp() {
    date -Iseconds 2>/dev/null || date -u +%Y-%m-%dT%H:%M:%SZ
}
EOF

display_billing() {

#!/bin/bash

check_security() {
    echo ""
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

        "${DATA_DIR}"
        "${CONFIG_DIR}"
        "${LOG_DIR}"
    )
    
    for dir in "${dirs[@]}"; do
        if [ -d "${dir}" ]; then
            local perms=$(stat -c "%a" "${dir}" 2>/dev/null || stat -f "%Lp" "${dir}" 2>/dev/null)
            if [ "${perms}" = "700" ] || [ "${perms}" = "755" ]; then
                echo "  ✓ ${dir}: ${perms} (secure)"
            else
                echo "  ⚠ ${dir}: ${perms} (recommend: 700)"
            fi
        fi
    done
    
    if [ -f "${BILLING_FILE}" ]; then
        local perms=$(stat -c "%a" "${BILLING_FILE}" 2>/dev/null || stat -f "%Lp" "${BILLING_FILE}" 2>/dev/null)
        if [ "${perms}" = "600" ]; then
            echo "  ✓ ${BILLING_FILE}: ${perms} (secure)"
        else
            echo "  ⚠ ${BILLING_FILE}: ${perms} (recommend: 600)"
        fi
    fi
    
    if [ -f "${BILLING_FILE}" ]; then
        if grep -q "password\|secret\|token\|key" "${BILLING_FILE}" 2>/dev/null; then
            echo "  ⚠ WARNING: Sensitive keywords detected in data file"

            echo "  ✓ No sensitive keywords detected"

# cursor-billing.ps1 - Windows PowerShell version
# Cursor Billing Display Tool

param(
    [Parameter(Position=0)]
    [ValidateSet('display','update','status','usage','help')]
    [string]$Command = 'display'
)

$BILLING_DIR = "$env:USERPROFILE\.cursor-billing"
$CONFIG_DIR = "$env:USERPROFILE\.config\cursor-billing"
$LOG_DIR = "$env:USERPROFILE\.cursor-billing\logs"
$BILLING_FILE = "$BILLING_DIR\billing-data.json"
$CONFIG_FILE = "$CONFIG_DIR\config.json"
$LOG_FILE = "$LOG_DIR\billing.log"

New-Item -ItemType Directory -Force -Path $BILLING_DIR | Out-Null
New-Item -ItemType Directory -Force -Path $CONFIG_DIR | Out-Null
New-Item -ItemType Directory -Force -Path $LOG_DIR | Out-Null

function Write-Log {
    param([string]$Message, [string]$Level = "info")
    $timestamp = (Get-Date -Format "yyyy-MM-ddTHH:mm:ssZ")
    "[${timestamp}] [${Level}] ${Message}" | Out-File -Append -FilePath $LOG_FILE
}

function Get-BillingData {
    if (Test-Path $BILLING_FILE) {
        Get-Content $BILLING_FILE -Raw | ConvertFrom-Json
    } else {
        $null
    }
}

function Save-BillingData {
    param($Data)
    $Data | ConvertTo-Json -Depth 10 | Set-Content $BILLING_FILE -Encoding UTF8
    Write-Log "Billing data updated"
}

function Display-Billing {
    $data = Get-BillingData
    
    if (-not $data) {
        Write-Host ""
        Write-Host "No billing data available. Run with 'update' first." -ForegroundColor Red

    Write-Host "  Billing Period:    $($data.billing_period)" -ForegroundColor White

function Update-Billing {
    Write-Host ""
    Write-Host "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━" -ForegroundColor Cyan
    Write-Host "  Update Billing Data" -ForegroundColor Cyan
    Write-Host "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━" -ForegroundColor Cyan
    Write-Host ""
    
    $plan = Read-Host "Plan (Pro/Free/Team) [Pro]"
    $period = Read-Host "Billing Period (Monthly/Annual) [Monthly]"
    $next_date = Read-Host "Next Billing Date (YYYY-MM-DD) [$((Get-Date).AddMonths(1).ToString('yyyy-MM-dd'))]"
    $amount = Read-Host "Amount (USD) [20.00]"
    $status = Read-Host "Status (active/inactive) [active]"
    $requests_used = Read-Host "Requests Used [0]"
    $requests_limit = Read-Host "Requests Limit [5000]"
    $tokens_used = Read-Host "Tokens Used [0]"
    $tokens_limit = Read-Host "Tokens Limit [5000000]"
    
    $data = @{
        plan = if ($plan) { $plan } else { "Pro" }
        billing_period = if ($period) { $period } else { "Monthly" }
        next_billing_date = if ($next_date) { $next_date } else { (Get-Date).AddMonths(1).ToString("yyyy-MM-dd") }
        amount = if ($amount) { [double]$amount } else { 20.00 }
        currency = "USD"
        status = if ($status) { $status } else { "active" }
        usage = @{
            requests_used = if ($requests_used) { [int]$requests_used } else { 0 }
            requests_limit = if ($requests_limit) { [int]$requests_limit } else { 5000 }
            tokens_used = if ($tokens_used) { [int]$tokens_used } else { 0 }
            tokens_limit = if ($tokens_limit) { [int]$tokens_limit } else { 5000000 }
        }
        payment_method = "•••• 4242"
        cached_at = (Get-Date -Format "yyyy-MM-ddTHH:mm:ssZ")
        source = "manual_update"
    }
    
    Save-BillingData $data
    Write-Host ""
    Write-Host "✓ Billing data updated successfully" -ForegroundColor Green
    Write-Host ""
}

function Check-Security {
    Write-Host ""
    Write-Host "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━" -ForegroundColor Cyan
    Write-Host "  Security Status" -ForegroundColor Cyan
    Write-Host "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━" -ForegroundColor Cyan
    
    $dirs = @($BILLING_DIR, $CONFIG_DIR, $LOG_DIR)
    foreach ($dir in $dirs) {
        if (Test-Path $dir) {
            Write-Host "  ✓ $dir" -ForegroundColor Green
        }
    }
    
    if (Test-Path $BILLING_FILE) {
        Write-Host "  ✓ Billing data file exists" -ForegroundColor Green
        $content = Get-Content $BILLING_FILE -Raw
        if ($content -match "password|secret|token|key") {
            Write-Host "  ⚠ WARNING: Sensitive keywords detected" -ForegroundColor Yellow
        } else {
            Write-Host "  ✓ No sensitive keywords detected" -ForegroundColor Green
        }
    }
    
    Write-Host "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━" -ForegroundColor Cyan
    Write-Host ""
}

COMMANDS:
  display     Show billing information

  help        Show this help message

EXAMPLES:
  .\cursor-billing.ps1 display
  .\cursor-billing.ps1 update

FILES:
  Data:     $BILLING_DIR

  Logs:     $LOG_DIR
"@
}
